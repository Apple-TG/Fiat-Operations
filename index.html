        // Update Time

        function updateTime() {

            const now = new Date();

            document.getElementById('current-time').innerText = now.toLocaleString();

        }

        // Address Verification

        function verifyAddress() {

            const addr = document.getElementById('trc-address').value.trim();

            const statusDiv = document.getElementById('address-status');

            

            if (!addr) {

                statusDiv.innerHTML = '';

                return;

            }


            if (addr.startsWith('T') && addr.length === 34) {

                statusDiv.innerHTML = '<span class="status-badge badge-success">Valid TRC20 Address</span>';

            } else {

                statusDiv.innerHTML = '<span class="status-badge badge-danger">Invalid TRC20 Address (Length or Prefix error)</span>';

            }

        }


        // Generate Audit Template

        function generateTRCTemplate() {

            const addr = document.getElementById('trc-address').value.trim();

            const amount = document.getElementById('payout-amount').value.trim();

            const record = document.getElementById('payout-record').value.trim();


            if (!addr || !amount) {

                alert('Please complete Address and Amount fields.');

                return;

            }


            const template = `[Payout Verification Request]

Address: ${addr}

Amount: ${amount} USDT

Status: Verified

Record Match: ${record ? 'Record Provided' : 'Pending Verification'}

Timestamp: ${new Date().toISOString()}`;


            document.getElementById('trc-output').innerText = template;

        }


        // Calculate Success Rate

        function calcSuccessRate() {

            const corridor = document.getElementById('corridor-select').value;

            const total = parseFloat(document.getElementById('total-txns').value);

            const success = parseFloat(document.getElementById('success-txns').value);


            if (!total || isNaN(success)) {

                alert('Please enter valid transaction numbers.');

                return;

            }


            const rate = ((success / total) * 100).toFixed(2);

            let status = rate >= 70 ? 'NORMAL' : 'LOW SUCCESS RATE ALERT';


            const report = `[Channel Performance Report]

Corridor: ${corridor}

Success Rate: ${rate}% (${success}/${total})

Status: ${status}

Generated: ${new Date().toLocaleTimeString()}`;


            document.getElementById('rate-output').innerText = report;

        }
